# 7. Advanced Python: Types and Simple Examples

## 1. Decorators

**a) Basic function decorator**
```python
def shout(func):
    def wrapper(*args, **kwargs):
        return func(*args, **kwargs).upper()
    return wrapper

@shout
def greet(name):
    return f"hello {name}"

print(greet("sam"))  # HELLO SAM
```

**b) Decorator preserving metadata (`functools.wraps`)**
```python
from functools import wraps

def log_call(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Calling {func.__name__}")
        return func(*args, **kwargs)
    return wrapper

@log_call
def add(a, b):
    return a + b

print(add(2, 3), add.__name__)  # add keeps its name
```

**c) Decorator with arguments**
```python
def repeat(times):
    def decorator(func):
        def wrapper(*args, **kwargs):
            for _ in range(times):
                func(*args, **kwargs)
        return wrapper
    return decorator

@repeat(3)
def hi():
    print("hi")

hi()
```

**d) Class-based decorator**
```python
class CountCalls:
    def __init__(self, func):
        self.func = func
        self.count = 0

    def __call__(self, *args, **kwargs):
        self.count += 1
        return self.func(*args, **kwargs)

@CountCalls
def ping():
    return "pong"

ping(); ping()
print(ping.count)  # 2
```

**e) Class decorator (decorating a class)**
```python
def add_hello(cls):
    cls.hello = lambda self: "hello"
    return cls

@add_hello
class Foo: ...

print(Foo().hello())
```

**f) Built-in decorators**
```python
class Circle:
    def __init__(self, r): self._r = r

    @property
    def area(self): return 3.14 * self._r ** 2

    @staticmethod
    def unit(): return 1

    @classmethod
    def from_diameter(cls, d): return cls(d / 2)
```

## 2. Context Managers

**a) Built-in `with`**
```python
with open("file.txt", "w") as f:
    f.write("hi")  # auto-closed
```

**b) Class-based (`__enter__` / `__exit__`)**
```python
class Timer:
    def __enter__(self):
        import time
        self.start = time.perf_counter()
        return self

    def __exit__(self, exc_type, exc, tb):
        import time
        print(f"{time.perf_counter() - self.start:.4f}s")
        return False  # don't suppress exceptions

with Timer():
    sum(range(1_000_000))
```

**c) Generator-based (`contextlib.contextmanager`)**
```python
from contextlib import contextmanager

@contextmanager
def tag(name):
    print(f"<{name}>")
    try:
        yield
    finally:
        print(f"</{name}>")

with tag("p"):
    print("text")
```

**d) `contextlib.suppress`**
```python
from contextlib import suppress
import os

with suppress(FileNotFoundError):
    os.remove("missing.txt")
```

**e) `ExitStack` (dynamic number of contexts)**
```python
from contextlib import ExitStack

with ExitStack() as stack:
    files = [stack.enter_context(open(f)) for f in ["a.txt", "b.txt"]]
```

**f) Async context manager**
```python
class AsyncRes:
    async def __aenter__(self): return self
    async def __aexit__(self, *exc): return False

# async with AsyncRes() as r: ...
```

## 3. Iterators and Generators (Advanced)

**a) Custom iterator class**
```python
class Countdown:
    def __init__(self, n): self.n = n
    def __iter__(self): return self
    def __next__(self):
        if self.n <= 0:
            raise StopIteration
        self.n -= 1
        return self.n + 1

print(list(Countdown(3)))  # [3, 2, 1]
```

**b) Generator function**
```python
def squares(n):
    for i in range(n):
        yield i * i

print(list(squares(4)))
```

**c) Generator expression**
```python
gen = (x * 2 for x in range(5))
print(sum(gen))
```

**d) `yield from` (delegation)**
```python
def chain(*iterables):
    for it in iterables:
        yield from it

print(list(chain([1, 2], "ab")))
```

**e) Two-way generator (`send`)**
```python
def accumulator():
    total = 0
    while True:
        value = yield total
        total += value

acc = accumulator()
next(acc)
print(acc.send(5), acc.send(10))  # 5 15
```

**f) Infinite generator with `itertools`**
```python
from itertools import count, islice

print(list(islice(count(10, 5), 4)))  # [10, 15, 20, 25]
```

**g) Iterable via `__iter__` using a generator**
```python
class Bag:
    def __init__(self, *items): self.items = items
    def __iter__(self):
        yield from self.items

print(list(Bag(1, 2, 3)))
```

**h) Async generator**
```python
async def ticker(n):
    for i in range(n):
        yield i

# async for x in ticker(3): ...
```

## 4. Type Hints

**a) Basic annotations**
```python
def add(a: int, b: int) -> int:
    return a + b

name: str = "Sam"
```

**b) Collections (built-in generics, Python 3.9+)**
```python
nums: list[int] = [1, 2]
ages: dict[str, int] = {"a": 1}
point: tuple[int, int] = (1, 2)
```

**c) Optional and Union (3.10+ syntax)**
```python
def find(id: int) -> str | None:
    return None
```

**d) `Callable`**
```python
from collections.abc import Callable

def apply(f: Callable[[int], int], x: int) -> int:
    return f(x)
```

**e) `TypeVar` / Generics**
```python
from typing import TypeVar

T = TypeVar("T")

def first(items: list[T]) -> T:
    return items[0]
```

**f) New generic syntax (3.12+)**
```python
def first[T](items: list[T]) -> T:
    return items[0]
```

**g) `TypedDict`**
```python
from typing import TypedDict

class User(TypedDict):
    name: str
    age: int
```

**h) `Protocol` (structural typing)**
```python
from typing import Protocol

class Speaker(Protocol):
    def speak(self) -> str: ...

class Dog:
    def speak(self) -> str: return "woof"

def talk(s: Speaker) -> str:
    return s.speak()
```

**i) `Literal` and `Final`**
```python
from typing import Literal, Final

mode: Literal["r", "w"] = "r"
MAX: Final = 10
```

**j) Dataclass with hints**
```python
from dataclasses import dataclass

@dataclass
class Point:
    x: int
    y: int
```

Check with a static checker: `mypy file.py` or `pyright file.py`. Hints aren't enforced at runtime.

## 5. Metaclasses

**a) `type()` creating a class dynamically**
```python
Dog = type("Dog", (), {"speak": lambda self: "woof"})
print(Dog().speak())
```

**b) Custom metaclass**
```python
class Meta(type):
    def __new__(mcs, name, bases, attrs):
        print(f"Creating {name}")
        return super().__new__(mcs, name, bases, attrs)

class Foo(metaclass=Meta): ...
```

**c) Enforcing rules (require an attribute)**
```python
class RequireId(type):
    def __new__(mcs, name, bases, attrs):
        if name != "Base" and "id" not in attrs:
            raise TypeError(f"{name} needs an 'id'")
        return super().__new__(mcs, name, bases, attrs)

class Base(metaclass=RequireId): ...
class Good(Base):
    id = 1
```

**d) Singleton via metaclass (`__call__`)**
```python
class Singleton(type):
    _instances = {}
    def __call__(cls, *args, **kwargs):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*args, **kwargs)
        return cls._instances[cls]

class Config(metaclass=Singleton): ...
print(Config() is Config())  # True
```

**e) Auto-registry via `__init_subclass__` (simpler alternative)**
```python
class Plugin:
    registry = []
    def __init_subclass__(cls, **kwargs):
        super().__init_subclass__(**kwargs)
        Plugin.registry.append(cls)

class A(Plugin): ...
class B(Plugin): ...
print(Plugin.registry)
```

**Tip:** Prefer `__init_subclass__` or class decorators over metaclasses when possible, since they are simpler.
