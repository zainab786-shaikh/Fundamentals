# 3. Python Object-Oriented Programming: Examples

## 3.1 Classes, Objects, `__init__`, Instance vs. Class Attributes
```python
class Dog:
    species = "Canis familiaris"      # class attribute (shared)

    def __init__(self, name, age):
        self.name = name              # instance attributes
        self.age = age

    def bark(self):
        return f"{self.name} says woof"

d1 = Dog("Rex", 3)
d2 = Dog("Fido", 5)
print(d1.bark())
print(d1.species, d2.species)         # shared
d1.name = "Max"                       # affects only d1
Dog.species = "Dog"                   # affects all instances
```

## 3.2 Four Pillars

**Encapsulation** (convention-based privacy)
```python
class Account:
    def __init__(self, balance):
        self._balance = balance       # _ = protected by convention
        self.__pin = "1234"           # __ = name-mangled (_Account__pin)

    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("Amount must be positive")
        self._balance += amount

    def get_balance(self):
        return self._balance
```

**Inheritance**
```python
class Animal:
    def __init__(self, name):
        self.name = name
    def speak(self):
        return "..."

class Cat(Animal):
    def __init__(self, name, indoor):
        super().__init__(name)        # call parent
        self.indoor = indoor
    def speak(self):                  # override
        return "Meow"

print(Cat("Tom", True).speak())
print(isinstance(Cat("Tom", True), Animal))   # True
```

**Polymorphism** (duck typing)
```python
class Duck:
    def speak(self): return "Quack"
class Robot:
    def speak(self): return "Beep"

for obj in [Duck(), Robot(), Cat("Tom", True)]:
    print(obj.speak())                # same call, different behavior
```

**Abstraction**
```python
from abc import ABC, abstractmethod

class Shape(ABC):
    @abstractmethod
    def area(self): ...

class Square(Shape):
    def __init__(self, side): self.side = side
    def area(self): return self.side ** 2

# Shape() raises TypeError; Square(2).area() -> 4
```

## 3.3 Dunder (Magic) Methods
```python
class Vector:
    def __init__(self, x, y):
        self.x, self.y = x, y
    def __repr__(self):               # developer-friendly
        return f"Vector({self.x}, {self.y})"
    def __str__(self):                # user-friendly
        return f"({self.x}, {self.y})"
    def __add__(self, other):         # +
        return Vector(self.x + other.x, self.y + other.y)
    def __eq__(self, other):          # ==
        return (self.x, self.y) == (other.x, other.y)
    def __len__(self):                # len()
        return 2
    def __getitem__(self, i):         # indexing
        return (self.x, self.y)[i]
    def __iter__(self):               # iteration
        yield self.x
        yield self.y

v = Vector(1, 2) + Vector(3, 4)
print(v, list(v), v[0], len(v))
```

## 3.4 `@property`, `@classmethod`, `@staticmethod`
```python
class Temperature:
    def __init__(self, celsius):
        self._c = celsius

    @property
    def celsius(self):                # getter
        return self._c

    @celsius.setter
    def celsius(self, value):         # setter with validation
        if value < -273.15:
            raise ValueError("Below absolute zero")
        self._c = value

    @property
    def fahrenheit(self):             # computed, read-only
        return self._c * 9 / 5 + 32

    @classmethod
    def from_fahrenheit(cls, f):      # alternative constructor
        return cls((f - 32) * 5 / 9)

    @staticmethod
    def is_valid(c):                  # no self/cls needed
        return c >= -273.15

t = Temperature.from_fahrenheit(212)
print(t.celsius, t.fahrenheit, Temperature.is_valid(-300))
```

## 3.5 ABCs, `dataclasses`, Mixins

**ABC with a shared method**
```python
from abc import ABC, abstractmethod

class Storage(ABC):
    @abstractmethod
    def save(self, data): ...
    def save_all(self, items):        # concrete method
        for i in items:
            self.save(i)
```

**`dataclasses`**
```python
from dataclasses import dataclass, field

@dataclass(frozen=True, order=True)
class Item:
    name: str
    price: float = 0.0
    tags: list[str] = field(default_factory=list)

a = Item("pen", 1.5)
print(a)                              # auto __repr__
print(a == Item("pen", 1.5))          # auto __eq__
```

**Mixin**
```python
import json

class JsonMixin:
    def to_json(self):
        return json.dumps(self.__dict__)

class User(JsonMixin):
    def __init__(self, name):
        self.name = name

print(User("Ada").to_json())          # {"name": "Ada"}
```

## 3.6 Composition vs. Inheritance
```python
# Inheritance: "is-a" (tight coupling)
class Engine:
    def start(self): return "Engine on"

class CarInherit(Engine):
    pass

# Composition: "has-a" (preferred, flexible)
class Car:
    def __init__(self, engine):
        self.engine = engine          # can swap any engine
    def start(self):
        return self.engine.start()

print(Car(Engine()).start())
```

## 3.7 Design Patterns and SOLID

**Singleton** (module-level instance is the simplest approach)
```python
class _Config:
    def __init__(self):
        self.debug = False

config = _Config()                    # import and reuse this one instance
```

**Factory**
```python
def make_shape(kind, size):
    shapes = {"square": Square}       # from 3.2
    return shapes[kind](size)
```

**Observer**
```python
class Subject:
    def __init__(self):
        self._observers = []
    def subscribe(self, fn):
        self._observers.append(fn)
    def notify(self, event):
        for fn in self._observers:
            fn(event)

s = Subject()
s.subscribe(lambda e: print("got", e))
s.notify("updated")
```

**Strategy** (functions as strategies)
```python
def apply_discount(price, strategy):
    return strategy(price)

print(apply_discount(100, lambda p: p * 0.9))
```

**SOLID, one-line examples**
```python
# S - Single responsibility: separate concerns
class Invoice: ...
class InvoicePrinter:
    def print(self, invoice): ...

# O - Open/closed: extend via new classes, not edits
class Circle(Shape):
    def __init__(self, r): self.r = r
    def area(self): return 3.14159 * self.r ** 2

# L - Liskov: subclasses must be usable wherever the parent is
def total_area(shapes: list[Shape]):
    return sum(s.area() for s in shapes)

# I - Interface segregation: small, focused interfaces
from typing import Protocol
class Readable(Protocol):
    def read(self) -> str: ...

# D - Dependency inversion: depend on abstractions
class Notifier:
    def __init__(self, sender: Storage):   # injected, not hardcoded
        self.sender = sender
```
