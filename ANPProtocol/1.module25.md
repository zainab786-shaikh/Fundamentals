Build an ANP Orchestrator Class
class ANPOrchestrator:

    def discover(self, capability):

        return [

            a for a in agents

            if capability in
               a["capabilities"]
        ]

    def route(self, candidates):

        return max(

            candidates,

            key=lambda a:
            a["trust_score"]
        )

    def execute(
        self,
        capability
    ):

        candidates = (
            self.discover(
                capability
            )
        )

        agent = self.route(
            candidates
        )

        return {

            "selected":
            agent["did"]
        }


Usage:

orchestrator = ANPOrchestrator()

result = orchestrator.execute(
    "weather_lookup"
)

print(result)


Output:

{
    "selected":
    "did:wba:weather.com"
}
