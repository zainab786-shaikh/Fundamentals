Step 2: Discovery Server
from fastapi import FastAPI

app = FastAPI()

registry = []

Register Endpoint
@app.post("/register")
def register(agent: dict):

    registry.append(agent)

    return {
        "status":
        "registered"
    }

Discovery Endpoint
@app.get("/discover/{capability}")
def discover(capability: str):

    return [

        agent

        for agent in registry

        if capability in
           agent["capabilities"]
    ]

Select best agent.

def select_best(agents):

    return max(

        agents,

        key=lambda a:
        a["trust_score"]
    )

Step 8: Weather Agent Service

Create a FastAPI service.

from fastapi import FastAPI

app = FastAPI()

Weather Endpoint
@app.post("/weather")
def weather(message: dict):

    city = (
        message["payload"]["city"]
    )

    return {

        "message_id":
        "MSG-002",

        "correlation_id":
        message["message_id"],

        "payload": {

            "weather":
            f"Sunny in {city}"
        }
    }
