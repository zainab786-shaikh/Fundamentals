Better Discovery Function
def discover(capability):

    matches = []

    for agent in registry:

        if capability in (
            agent["capabilities"]
        ):

            matches.append(agent)

    return matches

Discovery by DID

Sometimes you already know the identity.

def find_by_did(did):

    for agent in registry:

        if agent["did"] == did:

            return agent

required = {

    "weather_lookup",

    "forecast"
}

def discover(required):

    results = []

    for agent in registry:

        caps = set(
            agent["capabilities"]
        )

        if required.issubset(caps):

            results.append(agent)

    return results

Building a Discovery Service Class
class DiscoveryService:

    def __init__(self):

        self.agents = []

    def register(self, agent):

        self.agents.append(agent)

    def discover(
        self,
        capability
    ):

        return [

            agent

            for agent
            in self.agents

            if capability in
               agent["capabilities"]
        ]


Usage:

service = DiscoveryService()

service.register(
    weather_agent
)

service.register(
    finance_agent
)

print(
    service.discover(
        "weather_lookup"
    )
)
