## 2. Async SQLAlchemy Setup

Install an asynchronous PostgreSQL driver:

```bash name=install-async-db.sh
pip install sqlalchemy asyncpg
```

```python name=app/database_async.py
import os
from collections.abc import AsyncGenerator

from sqlalchemy.ext.asyncio import (
    AsyncSession,
    async_sessionmaker,
    create_async_engine,
)

database_url = os.environ["DATABASE_URL"]

engine = create_async_engine(
    database_url,
    pool_pre_ping=True,
)

AsyncSessionLocal = async_sessionmaker(
    engine,
    expire_on_commit=False,
)


async def get_async_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        yield session
```

Example environment value:

```text name=.env.example
DATABASE_URL=postgresql+asyncpg://app_user:password@localhost:5432/app_db
```

Store the real connection string in a secret manager or environment variable.

## 3. Async Database Query

```python name=app/routers/users_async.py
from typing import Annotated

from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.database_async import get_async_db
from app.models.user import User

router = APIRouter(prefix="/users", tags=["Users"])

AsyncDatabase = Annotated[AsyncSession, Depends(get_async_db)]


@router.get("/{user_id}")
async def get_user(
    user_id: int,
    db: AsyncDatabase,
) -> dict[str, object]:
    result = await db.execute(
        select(User).where(User.id == user_id),
    )
    user = result.scalar_one_or_none()

    if user is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="User not found",
        )

    return {"id": user.id, "email": user.email}
```

Await asynchronous database operations. SQLAlchemy parameterizes values to prevent SQL injection.

## 4. Background Tasks

```python name=app/routers/notifications.py
import logging

from fastapi import APIRouter, BackgroundTasks, status

router = APIRouter()
logger = logging.getLogger(__name__)


def send_welcome_message(email: str) -> None:
    logger.info("Sending welcome message")
    # Call the email provider here without logging the address or credentials.


@router.post("/notifications", status_code=status.HTTP_202_ACCEPTED)
def create_notification(
    email: str,
    background_tasks: BackgroundTasks,
) -> dict[str, str]:
    background_tasks.add_task(send_welcome_message, email)
    return {"status": "accepted"}
```

Background tasks run after the response. Use a task queue for important, long-running, or retryable work.

## 5. Secure File Uploads

```python name=app/routers/uploads.py
from pathlib import Path
from uuid import uuid4

from fastapi import APIRouter, HTTPException, UploadFile, status

router = APIRouter()

UPLOAD_DIRECTORY = Path("uploads").resolve()
UPLOAD_DIRECTORY.mkdir(parents=True, exist_ok=True)

ALLOWED_CONTENT_TYPES = {"image/jpeg", "image/png"}
MAX_FILE_SIZE = 2 * 1024 * 1024


@router.post("/uploads", status_code=status.HTTP_201_CREATED)
async def upload_image(file: UploadFile) -> dict[str, str]:
    if file.content_type not in ALLOWED_CONTENT_TYPES:
        raise HTTPException(415, "Only JPEG and PNG files are allowed")

    content = await file.read(MAX_FILE_SIZE + 1)

    if len(content) > MAX_FILE_SIZE:
        raise HTTPException(413, "File is too large")

    extension = ".jpg" if file.content_type == "image/jpeg" else ".png"
    safe_name = f"{uuid4()}{extension}"
    destination = UPLOAD_DIRECTORY / safe_name

    destination.write_bytes(content)

    return {"file_name": safe_name}
```

Generate the filename on the server to prevent path traversal. Also verify the actual file signature and scan uploads for malware in production.

## 6. WebSockets

```python name=app/routers/websocket.py
from fastapi import APIRouter, WebSocket, WebSocketDisconnect

router = APIRouter()


@router.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket) -> None:
    token = websocket.query_params.get("token")

    if token != "temporary-example-token":
        await websocket.close(code=1008)
        return

    await websocket.accept()

    try:
        while True:
            message = await websocket.receive_text()
            await websocket.send_json({"message": message})
    except WebSocketDisconnect:
        pass
```

WebSockets provide two-way real-time communication. In production, validate a short-lived token securely instead of using a hardcoded value.

## 7. Custom Middleware

```python name=app/main.py
from time import perf_counter

from fastapi import FastAPI, Request, Response

app = FastAPI()


@app.middleware("http")
async def add_processing_time(
    request: Request,
    call_next,
) -> Response:
    started_at = perf_counter()
    response = await call_next(request)
    duration = perf_counter() - started_at

    response.headers["X-Process-Time"] = f"{duration:.4f}"
    return response
```

Middleware runs for every HTTP request. It is useful for logging, security headers, request IDs, and performance measurement.

## 11. Call an External API

```python name=app/services/external_api.py
import httpx


async def get_exchange_rate(currency: str) -> dict[str, object]:
    allowed_currencies = {"EUR", "GBP", "JPY"}

    if currency not in allowed_currencies:
        raise ValueError("Unsupported currency")

    url = f"https://api.example.com/rates/{currency}"

    timeout = httpx.Timeout(5.0, connect=2.0)

    async with httpx.AsyncClient(timeout=timeout) as client:
        response = await client.get(url)
        response.raise_for_status()
        return response.json()
```

Use HTTPS, strict allow-lists, and timeouts. Never accept an unrestricted URL from the client because it could enable server-side request forgery.

## 12. Retry Temporary External Failures

```python name=app/services/retry.py
import asyncio

import httpx


async def fetch_with_retry(url: str) -> httpx.Response:
    allowed_url = "https://api.example.com/status"

    if url != allowed_url:
        raise ValueError("URL is not allowed")

    for attempt in range(3):
        try:
            async with httpx.AsyncClient(timeout=5.0) as client:
                response = await client.get(url)
                response.raise_for_status()
                return response
        except (
            httpx.ConnectError,
            httpx.TimeoutException,
        ):
            if attempt == 2:
                raise

            await asyncio.sleep(2**attempt)

    raise RuntimeError("Request failed")
```

Retry only temporary failures and use increasing delays. Avoid retrying validation errors or unsafe operations that could be applied twice.

## 13. Application Lifespan

```python name=app/main.py
from collections.abc import AsyncIterator
from contextlib import asynccontextmanager

from fastapi import FastAPI


@asynccontextmanager
async def lifespan(app: FastAPI) -> AsyncIterator[None]:
    print("Application starting")
    yield
    print("Application stopping")


app = FastAPI(lifespan=lifespan)
```

Lifespan logic initializes and closes shared resources such as HTTP clients, cache connections, and database pools.

## 14. Rate Limiting Concept

```python name=app/services/rate_limit.py
from collections import defaultdict
from time import monotonic

requests_by_client: dict[str, list[float]] = defaultdict(list)


def is_allowed(client_id: str) -> bool:
    now = monotonic()
    window_start = now - 60

    recent = [
        timestamp
        for timestamp in requests_by_client[client_id]
        if timestamp > window_start
    ]

    if len(recent) >= 10:
        return False

    recent.append(now)
    requests_by_client[client_id] = recent
    return True
```

This demonstrates the principle only. For production, use a trusted reverse proxy, API gateway, or Redis-backed limiter so limits work across multiple processes.

## Key Guidelines

- Use async code only with async-compatible libraries.
- Validate file type, size, and filename.
- Authenticate WebSocket connections.
- Apply strict URL allow-lists to external requests.
- Set network timeouts and limited retries.
- Keep heavy jobs outside the API process.
- Never cache secrets or authorization responses carelessly.
- Use shared infrastructure for production caching and rate limiting.
