
# Topic 10: Testing FastAPI with `pytest`

Testing confirms that routes, validation, authentication, permissions, and database operations work correctly.

## 1. Install Testing Packages

```bash name=install-testing.sh
pip install pytest httpx pytest-cov
```

- `pytest`: Test framework
- `httpx`: Used by FastAPI’s test client
- `pytest-cov`: Measures test coverage

## 4. Create an Isolated Test Database

```python name=tests/conftest.py
from collections.abc import Generator

import pytest
from fastapi.testclient import TestClient
from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker
from sqlalchemy.pool import StaticPool

from app.database import Base
from app.dependencies import get_db
from app.main import app

test_engine = create_engine(
    "sqlite://",
    connect_args={"check_same_thread": False},
    poolclass=StaticPool,
)

TestSessionLocal = sessionmaker(
    bind=test_engine,
    autoflush=False,
    expire_on_commit=False,
)


@pytest.fixture
def db() -> Generator[Session, None, None]:
    Base.metadata.create_all(bind=test_engine)

    with TestSessionLocal() as session:
        yield session

    Base.metadata.drop_all(bind=test_engine)


@pytest.fixture
def client(db: Session) -> Generator[TestClient, None, None]:
    def override_get_db() -> Generator[Session, None, None]:
        yield db

    app.dependency_overrides[get_db] = override_get_db

    with TestClient(app) as test_client:
        yield test_client

    app.dependency_overrides.clear()
```

Each test receives an isolated in-memory database. Dependency overriding prevents tests from touching development or production data.

## 9. Test Duplicate Records

```python name=tests/test_users.py
from fastapi.testclient import TestClient


def test_duplicate_email_returns_conflict(
    client: TestClient,
) -> None:
    payload = {
        "name": "Alice",
        "email": "alice@example.com",
    }

    first_response = client.post("/users", json=payload)
    second_response = client.post("/users", json=payload)

    assert first_response.status_code == 201
    assert second_response.status_code == 409
    assert second_response.json() == {
        "detail": "Email is already registered",
    }
```

The API should reject duplicate unique values with `409 Conflict`.

## 10. Test Partial Updates

```python name=tests/test_users.py
from fastapi.testclient import TestClient


def test_update_user_name(client: TestClient) -> None:

        "/users",

            "name": "Alice",
            "email": "alice@example.com",
        },
    )

    response = client.patch(
        f"/users/{user_id}",

    assert body["name"] == "Alice Smith"
    assert body["email"] == "alice@example.com"
```

This confirms that `PATCH` changes only the supplied fields.

## 11. Test Deletion

```python name=tests/test_users.py
from fastapi.testclient import TestClient


def test_delete_user(client: TestClient) -> None:
    created = client.post(
        "/users",
        json={
            "name": "Alice",
            "email": "alice@example.com",
        },
    )
    user_id = created.json()["id"]

    delete_response = client.delete(f"/users/{user_id}")
    get_response = client.get(f"/users/{user_id}")

    assert delete_response.status_code == 204
    assert delete_response.content == b""
    assert get_response.status_code == 404
```

The record should be unavailable after deletion.

## 12. Test Registration Security

```python name=tests/test_security.py
from fastapi.testclient import TestClient
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.user import User


def test_registration_hashes_password(
    client: TestClient,
    db: Session,
) -> None:
    password = "SecurePassword!123"

    response = client.post(
        "/auth/register",
        json={
            "email": "alice@example.com",
            "password": password,
        },
    )

    user = db.scalar(
        select(User).where(User.email == "alice@example.com"),
    )

    assert response.status_code == 201
    assert user is not None
    assert user.password_hash != password
    assert password not in user.password_hash
```

This verifies that the original password is never stored directly.

## 15. Test with an Access Token

```python name=tests/test_security.py
from fastapi.testclient import TestClient


def test_authenticated_user_can_view_profile(
    client: TestClient,
) -> None:
    client.post(
        "/auth/register",
        json={
            "email": "alice@example.com",
            "password": "SecurePassword!123",
        },
    )

    login_response = client.post(
        "/auth/token",
        data={
            "username": "alice@example.com",
            "password": "SecurePassword!123",
        },
    )
    token = login_response.json()["access_token"]

    response = client.get(
        "/profile",
        headers={"Authorization": f"Bearer {token}"},
    )

    assert response.status_code == 200
    assert response.json()["email"] == "alice@example.com"
```

The token is sent through the `Authorization` header using the Bearer scheme.

## 17. Unit-Test a Service

```python name=tests/test_user_service.py
from unittest.mock import Mock

from sqlalchemy.orm import Session

from app.repositories.user_repository import UserRepository
from app.schemas.user import UserCreate
from app.services.user_service import UserService


def test_service_creates_user() -> None:
    repository = Mock(spec=UserRepository)
    database = Mock(spec=Session)
    expected_user = Mock(id=1, name="Alice")

    repository.get_by_email.return_value = None
    repository.create.return_value = expected_user

    service = UserService(repository)
    result = service.create_user(
        database,
        UserCreate(
            name="Alice",
            email="alice@example.com",
        ),
    )

    assert result == expected_user
    repository.create.assert_called_once()
    database.commit.assert_called_once()
```

A unit test isolates business logic by replacing the repository and database with mocks.

## 18. Run the Tests

```bash name=run-tests.sh
pytest -v
```

Run tests with coverage:

```bash name=run-coverage.sh
pytest --cov=app --cov-report=term-missing
```

Coverage identifies untested lines, but high coverage does not guarantee correct behavior.

## Important Test Cases

Test at least:

- Successful CRUD operations
- Invalid request data
- Missing records
- Duplicate values
- Database rollback behavior
- Invalid and expired tokens
- Anonymous access
- Insufficient permissions
- Disabled users
- Pagination limits
- Sensitive fields excluded from responses
