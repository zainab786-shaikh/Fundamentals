## 4.1 Complexity Analysis (Big O)
```python
def first(items):            # O(1): constant
    return items[0]

def contains(items, x):      # O(n): linear
    for i in items:
        if i == x:
            return True
    return False

def has_duplicate_pair(items):   # O(n^2): quadratic
    for i in items:
        for j in items:
            if i is not j and i == j:
                return True
    return False

def halve(n):                # O(log n): logarithmic
    steps = 0
    while n > 1:
        n //= 2
        steps += 1
    return steps

# Quick timing check
import timeit
print(timeit.timeit("sorted(range(1000))", number=1000))
```

## 4.2 Searching and Sorting

**Linear search** O(n)
```python
def linear_search(items, target):
    for i, v in enumerate(items):
        if v == target:
            return i
    return -1
```

**Binary search** O(log n), requires sorted input
```python
def binary_search(items, target):
    lo, hi = 0, len(items) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if items[mid] == target:
            return mid
        if items[mid] < target:
            lo = mid + 1
        else:
            hi = mid - 1
    return -1

print(binary_search([1, 3, 5, 7, 9], 7))   # 3
```

**Bubble sort** O(n²)
```python
def bubble_sort(a):
    a = a[:]
    for i in range(len(a)):
        for j in range(len(a) - i - 1):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]
    return a
```

**Insertion sort** O(n²)
```python
def insertion_sort(a):
    a = a[:]
    for i in range(1, len(a)):
        key, j = a[i], i - 1
        while j >= 0 and a[j] > key:
            a[j + 1] = a[j]
            j -= 1
        a[j + 1] = key
    return a
```

**Merge sort** O(n log n)
```python
def merge_sort(a):
    if len(a) <= 1:
        return a
    mid = len(a) // 2
    left, right = merge_sort(a[:mid]), merge_sort(a[mid:])
    out, i, j = [], 0, 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            out.append(left[i]); i += 1
        else:
            out.append(right[j]); j += 1
    return out + left[i:] + right[j:]
```

**Quick sort** O(n log n) average
```python
def quick_sort(a):
    if len(a) <= 1:
        return a
    pivot = a[len(a) // 2]
    return (quick_sort([x for x in a if x < pivot])
            + [x for x in a if x == pivot]
            + quick_sort([x for x in a if x > pivot]))
```

## 4.3 Recursion and Backtracking

**Recursion**
```python
def factorial(n):
    return 1 if n <= 1 else n * factorial(n - 1)

def fib(n, memo={}):
    if n < 2:
        return n
    if n not in memo:
        memo[n] = fib(n - 1) + fib(n - 2)
    return memo[n]
```

**Backtracking: permutations**
```python
def permutations(items):
    result = []
    def backtrack(path, remaining):
        if not remaining:
            result.append(path[:])
            return
        for i in range(len(remaining)):
            path.append(remaining[i])                        # choose
            backtrack(path, remaining[:i] + remaining[i+1:]) # explore
            path.pop()                                       # undo
    backtrack([], items)
    return result

print(permutations([1, 2, 3]))
```

**Backtracking: subsets**
```python
def subsets(nums):
    res = []
    def go(start, path):
        res.append(path[:])
        for i in range(start, len(nums)):
            path.append(nums[i])
            go(i + 1, path)
            path.pop()
    go(0, [])
    return res
```

## 4.4 Divide and Conquer, Greedy, Dynamic Programming

**Divide and conquer: max subarray split**
```python
def power(x, n):             # O(log n)
    if n == 0:
        return 1
    half = power(x, n // 2)
    return half * half if n % 2 == 0 else half * half * x

print(power(2, 10))   # 1024
```

**Greedy: coin change (canonical coin systems)**
```python
def greedy_coins(amount, coins=(25, 10, 5, 1)):
    result = []
    for c in coins:
        while amount >= c:
            amount -= c
            result.append(c)
    return result

print(greedy_coins(63))   # [25, 25, 10, 1, 1, 1]
```

**Greedy: activity selection**
```python
def max_activities(intervals):
    intervals.sort(key=lambda x: x[1])      # by end time
    count, end = 0, float("-inf")
    for s, e in intervals:
        if s >= end:
            count += 1
            end = e
    return count
```

**Dynamic programming: top-down (memoization)**
```python
from functools import lru_cache

@lru_cache(maxsize=None)
def climb(n):
    return 1 if n <= 1 else climb(n - 1) + climb(n - 2)
```

**Dynamic programming: bottom-up (tabulation), min coins**
```python
def min_coins(coins, amount):
    dp = [0] + [float("inf")] * amount
    for a in range(1, amount + 1):
        for c in coins:
            if c <= a:
                dp[a] = min(dp[a], dp[a - c] + 1)
    return dp[amount] if dp[amount] != float("inf") else -1

print(min_coins([1, 5, 6], 10))   # 2
```

## 1. Divide and conquer

| Variant | Idea |
|---|---|
| Merge sort | Split, sort halves, merge |
| Quick sort | Partition around a pivot |
| Binary search | Halve the search space |
| Max subarray | Split, then combine left, right and crossing |
| Fast exponentiation | Square the base, halve the exponent |

```python name=divide_conquer.py
# Quick sort
def quick_sort(a):
    if len(a) <= 1:
        return a
    p = a[len(a) // 2]
    return (quick_sort([x for x in a if x < p])
            + [x for x in a if x == p]
            + quick_sort([x for x in a if x > p]))

# Binary search (iterative)
def binary_search(a, target):
    lo, hi = 0, len(a) - 1
    while lo <= hi:
        mid = (lo + hi) // 2
        if a[mid] == target: return mid
        if a[mid] < target: lo = mid + 1
        else: hi = mid - 1
    return -1

# Fast exponentiation
def power(x, n):
    if n == 0: return 1
    half = power(x, n // 2)
    return half * half * (x if n % 2 else 1)

# Max subarray (divide and conquer)
def max_subarray(a, lo=0, hi=None):
    hi = len(a) - 1 if hi is None else hi
    if lo == hi: return a[lo]
    mid = (lo + hi) // 2
    left_best, s = float("-inf"), 0
    for i in range(mid, lo - 1, -1):
        s += a[i]; left_best = max(left_best, s)
    right_best, s = float("-inf"), 0
    for i in range(mid + 1, hi + 1):
        s += a[i]; right_best = max(right_best, s)
    return max(max_subarray(a, lo, mid),
               max_subarray(a, mid + 1, hi),
               left_best + right_best)

print(quick_sort([3, 6, 1, 8, 2]))           # [1, 2, 3, 6, 8]
print(binary_search([1, 3, 5, 7], 5))        # 2
print(power(2, 10))                          # 1024
print(max_subarray([-2, 1, -3, 4, -1, 2]))   # 5
```

## 2. Greedy

| Variant | Idea |
|---|---|
| Activity selection | Pick the earliest finish time |
| Fractional knapsack | Pick the best value/weight ratio first |
| Huffman coding | Repeatedly merge the two lowest frequencies |
| Jump game | Track the farthest reachable index |
| Minimum platforms and meeting rooms | Sort and sweep |

```python name=greedy.py
import heapq

# Activity selection
def max_activities(intervals):
    count, end = 0, float("-inf")
    for s, e in sorted(intervals, key=lambda x: x[1]):
        if s >= end:
            count += 1; end = e
    return count

# Fractional knapsack
def fractional_knapsack(items, cap):  # items = (value, weight)
    total = 0.0
    for v, w in sorted(items, key=lambda x: x[0] / x[1], reverse=True):
        take = min(w, cap)
        total += v * take / w
        cap -= take
        if cap == 0: break
    return total

# Huffman: total cost of merging
def huffman_cost(freqs):
    heapq.heapify(freqs)
    cost = 0
    while len(freqs) > 1:
        m = heapq.heappop(freqs) + heapq.heappop(freqs)
        cost += m
        heapq.heappush(freqs, m)
    return cost

# Jump game
def can_jump(nums):
    far = 0
    for i, n in enumerate(nums):
        if i > far: return False
        far = max(far, i + n)
    return True

print(max_activities([(1, 3), (2, 4), (3, 5), (0, 6)]))        # 2
print(fractional_knapsack([(60, 10), (100, 20), (120, 30)], 50))  # 240.0
print(huffman_cost([5, 9, 12, 13, 16, 45]))                    # 224
print(can_jump([2, 3, 1, 1, 4]))                               # True
```

## 3. Dynamic programming

| Variant | Idea |
|---|---|
| Top-down (memoization) | Recursion plus a cache |
| Bottom-up (tabulation) | Fill a table iteratively |
| Space-optimized | Keep only the previous state |
| 1D DP | Climbing stairs, house robber |
| 2D DP | LCS, edit distance, grid paths |
| Knapsack (0/1) | Take or skip each item |

```python name=dp_types.py
from functools import cache

# Top-down
@cache
def climb(n):
    return n if n <= 2 else climb(n - 1) + climb(n - 2)

# Bottom-up
def climb_table(n):
    dp = [0, 1, 2] + [0] * max(0, n - 2)
    for i in range(3, n + 1):
        dp[i] = dp[i - 1] + dp[i - 2]
    return dp[n]

# Space-optimized
def climb_opt(n):
    a, b = 1, 2
    for _ in range(n - 2):
        a, b = b, a + b
    return b if n >= 2 else 1

# 1D: house robber
def rob(nums):
    prev = cur = 0
    for n in nums:
        prev, cur = cur, max(cur, prev + n)
    return cur

# 2D: longest common subsequence
def lcs(a, b):
    dp = [[0] * (len(b) + 1) for _ in range(len(a) + 1)]
    for i in range(1, len(a) + 1):
        for j in range(1, len(b) + 1):
            dp[i][j] = (dp[i-1][j-1] + 1 if a[i-1] == b[j-1]
                        else max(dp[i-1][j], dp[i][j-1]))
    return dp[-1][-1]

# 0/1 knapsack
def knapsack(weights, values, cap):
    dp = [0] * (cap + 1)
    for w, v in zip(weights, values):
        for c in range(cap, w - 1, -1):
            dp[c] = max(dp[c], dp[c - w] + v)
    return dp[cap]

print(climb(5), climb_table(5), climb_opt(5))  # 8 8 8
print(rob([2, 7, 9, 3, 1]))                    # 12
print(lcs("abcde", "ace"))                     # 3
print(knapsack([1, 3, 4], [15, 20, 30], 4))    # 35
```

## 4. Graph algorithms

| Variant | Idea |
|---|---|
| BFS (queue) | Level by level, shortest path in an unweighted graph |
| DFS (recursive) | Go deep, then backtrack |
| DFS (iterative, stack) | Avoids recursion limits |
| Dijkstra | Shortest path, non-negative weights |
| Bellman-Ford | Shortest path, handles negative weights |
| Topological sort | Order a DAG's dependencies |
| Union-Find | Connectivity and cycle detection |

```python name=graph_types.py
from collections import deque
import heapq

g = {"A": ["B", "C"], "B": ["D"], "C": ["D"], "D": []}

# BFS shortest path (unweighted)
def bfs_dist(start):
    dist, q = {start: 0}, deque([start])
    while q:
        n = q.popleft()
        for nb in g[n]:
            if nb not in dist:
                dist[nb] = dist[n] + 1; q.append(nb)
    return dist

# DFS recursive
def dfs_rec(n, seen=None):
    seen = set() if seen is None else seen
    seen.add(n)
    for nb in g[n]:
        if nb not in seen: dfs_rec(nb, seen)
    return seen

# DFS iterative
def dfs_iter(start):
    seen, stack = set(), [start]
    while stack:
        n = stack.pop()
        if n in seen: continue
        seen.add(n); stack.extend(g[n])
    return seen

# Topological sort (Kahn)
def topo(graph):
    indeg = {n: 0 for n in graph}
    for n in graph:
        for nb in graph[n]: indeg[nb] += 1
    q = deque(n for n in indeg if indeg[n] == 0)
    order = []
    while q:
        n = q.popleft(); order.append(n)
        for nb in graph[n]:
            indeg[nb] -= 1
            if indeg[nb] == 0: q.append(nb)
    return order

# Bellman-Ford (edges = (u, v, w))
def bellman_ford(nodes, edges, src):
    dist = {n: float("inf") for n in nodes}; dist[src] = 0
    for _ in range(len(nodes) - 1):
        for u, v, w in edges:
            if dist[u] + w < dist[v]: dist[v] = dist[u] + w
    return dist

# Union-Find
class DSU:
    def __init__(self, n): self.p = list(range(n))
    def find(self, x):
        while self.p[x] != x:
            self.p[x] = self.p[self.p[x]]; x = self.p[x]
        return x
    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb: return False  # already connected (cycle)
        self.p[ra] = rb; return True

print(bfs_dist("A"))                 # {'A': 0, 'B': 1, 'C': 1, 'D': 2}
print(dfs_rec("A"), dfs_iter("A"))
print(topo(g))                       # ['A', 'B', 'C', 'D']
print(bellman_ford("ABC", [("A","B",4), ("B","C",-2)], "A"))  # A:0, B:4, C:2
d = DSU(3); print(d.union(0, 1), d.union(0, 1))  # True False
```

## 5. String algorithms

| Variant | Idea |
|---|---|
| Two pointers | Palindrome, reverse |
| Hash map / counter | Anagram, first unique char |
| Sliding window | Longest substring without repeats |
| Pattern matching | Naive, KMP (previous message), Rabin-Karp |
| DP on strings | Edit distance |
| Stack-based | Balanced parentheses |

```python name=string_types.py
from collections import Counter

# Two pointers
def is_pal(s):
    i, j = 0, len(s) - 1
    while i < j:
        if s[i] != s[j]: return False
        i += 1; j -= 1
    return True

# Hash map: anagram
def is_anagram(a, b):
    return Counter(a) == Counter(b)

# Sliding window
def longest_unique(s):
    seen, left, best = {}, 0, 0
    for r, ch in enumerate(s):
        if ch in seen and seen[ch] >= left:
            left = seen[ch] + 1
        seen[ch] = r
        best = max(best, r - left + 1)
    return best

# Rabin-Karp (rolling hash)
def rabin_karp(text, pat, base=256, mod=1_000_003):
    n, m = len(text), len(pat)
    if m > n: return -1
    h = pow(base, m - 1, mod)
    ph = th = 0
    for i in range(m):
        ph = (ph * base + ord(pat[i])) % mod
        th = (th * base + ord(text[i])) % mod
    for i in range(n - m + 1):
        if ph == th and text[i:i+m] == pat: return i
        if i < n - m:
            th = ((th - ord(text[i]) * h) * base + ord(text[i+m])) % mod
    return -1

# Edit distance (DP)
def edit_distance(a, b):
    dp = list(range(len(b) + 1))
    for i in range(1, len(a) + 1):
        prev, dp[0] = dp[0], i
        for j in range(1, len(b) + 1):
            prev, dp[j] = dp[j], min(dp[j] + 1, dp[j-1] + 1,
                                     prev + (a[i-1] != b[j-1]))
    return dp[-1]

# Stack: balanced brackets
def balanced(s):
    pairs, st = {")": "(", "]": "[", "}": "{"}, []
    for c in s:
        if c in "([{": st.append(c)
        elif c in pairs:
            if not st or st.pop() != pairs[c]: return False
    return not st

print(is_pal("racecar"), is_anagram("listen", "silent"))  # True True
print(longest_unique("abcabcbb"))                         # 3
print(rabin_karp("hello world", "world"))                 # 6
print(edit_distance("kitten", "sitting"))                 # 3
print(balanced("{[()]}"))                                 # True
```
