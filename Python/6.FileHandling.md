# 6. File Handling and Serialization

Python examples, one per variant. Each security note explains the risk it mitigates.

## 6.1 File I/O: different ways to read

```python
# 1. Read the whole file as a string
with open("data.txt", encoding="utf-8") as f:
    text = f.read()

# 2. Read all lines into a list
with open("data.txt", encoding="utf-8") as f:
    lines = f.readlines()

# 3. Iterate line by line (memory-efficient for large files)
with open("data.txt", encoding="utf-8") as f:
    for line in f:
        print(line.rstrip())

# 4. Read in fixed-size chunks (huge or binary files)
with open("big.bin", "rb") as f:
    while chunk := f.read(8192):
        process(chunk)

# 5. Read a single line
with open("data.txt", encoding="utf-8") as f:
    first = f.readline()
```

## 6.2 File I/O: different ways to write

```python
# 1. Write (overwrite)
with open("out.txt", "w", encoding="utf-8") as f:
    f.write("hello\n")

# 2. Append
with open("out.txt", "a", encoding="utf-8") as f:
    f.write("more\n")

# 3. Write many lines
with open("out.txt", "w", encoding="utf-8") as f:
    f.writelines(f"{i}\n" for i in range(3))

# 4. Exclusive create (fails if the file exists, so it avoids overwriting)
with open("new.txt", "x", encoding="utf-8") as f:
    f.write("only if new")

# 5. Binary write
with open("img.bin", "wb") as f:
    f.write(b"\x00\x01")

# 6. print() to a file
with open("out.txt", "w", encoding="utf-8") as f:
    print("hello", file=f)
```

**Atomic write.** This prevents corrupted or half-written files:

```python
import os, tempfile

def atomic_write(path: str, data: str) -> None:
    dir_ = os.path.dirname(os.path.abspath(path))
    with tempfile.NamedTemporaryFile("w", dir=dir_, delete=False, encoding="utf-8") as tmp:
        tmp.write(data)
    os.replace(tmp.name, path)  # atomic on the same filesystem
```

## 6.3 `pathlib`

```python
from pathlib import Path

p = Path("data") / "notes.txt"          # 1. Build paths with /

p.exists(); p.is_file(); p.is_dir()      # 2. Checks

p.name; p.stem; p.suffix; p.parent       # 3. Components

text = p.read_text(encoding="utf-8")     # 4. Quick read/write
p.write_text("hi", encoding="utf-8")
data = p.read_bytes()

Path("a/b/c").mkdir(parents=True, exist_ok=True)   # 5. Create directories

for f in Path(".").glob("*.py"):         # 6. Glob
    print(f)
for f in Path(".").rglob("*.json"):      # 7. Recursive glob
    print(f)

for child in Path(".").iterdir():        # 8. List a directory
    print(child)

p.rename("new.txt"); p.unlink(missing_ok=True)   # 9. Rename and delete

p.resolve(); Path.home(); Path.cwd()     # 10. Absolute and special paths
```

**Security note: path traversal.** This blocks inputs like `../../etc/passwd`:

```python
from pathlib import Path

BASE = Path("/srv/uploads").resolve()

def safe_path(user_input: str) -> Path:
    target = (BASE / user_input).resolve()
    if not target.is_relative_to(BASE):   # Python 3.9+
        raise ValueError("Path traversal attempt blocked")
    return target
```

## 6.4 JSON

```python
import json

# 1. Object -> string
s = json.dumps({"name": "Ana", "age": 30})

# 2. String -> object
obj = json.loads('{"name": "Ana"}')

# 3. Write to a file
with open("d.json", "w", encoding="utf-8") as f:
    json.dump({"a": 1}, f)

# 4. Read from a file
with open("d.json", encoding="utf-8") as f:
    obj = json.load(f)

# 5. Pretty-print and sort keys
print(json.dumps(obj, indent=2, sort_keys=True))

# 6. Keep non-ASCII characters readable
json.dumps({"city": "München"}, ensure_ascii=False)

# 7. Custom types via default=
from datetime import datetime
json.dumps({"t": datetime.now()}, default=str)

# 8. Error handling
try:
    json.loads("{bad json}")
except json.JSONDecodeError as e:
    print(f"Invalid JSON at line {e.lineno}: {e.msg}")

# 9. JSON Lines (one object per line, good for logs and streaming)
with open("log.jsonl", "a", encoding="utf-8") as f:
    f.write(json.dumps({"event": "login"}) + "\n")
```

## 6.5 CSV

```python
import csv

# 1. Read as lists
with open("d.csv", newline="", encoding="utf-8") as f:
    for row in csv.reader(f):
        print(row)

# 2. Read as dicts (uses the header row)
with open("d.csv", newline="", encoding="utf-8") as f:
    for row in csv.DictReader(f):
        print(row["name"])

# 3. Write lists
with open("o.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.writer(f)
    w.writerow(["name", "age"])
    w.writerows([["Ana", 30], ["Bo", 25]])

# 4. Write dicts
with open("o.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.DictWriter(f, fieldnames=["name", "age"])
    w.writeheader()
    w.writerow({"name": "Ana", "age": 30})

# 5. Custom delimiter
with open("d.tsv", newline="", encoding="utf-8") as f:
    rows = list(csv.reader(f, delimiter="\t"))
```

**Security note: CSV injection.** Cells starting with `=`, `+`, `-` or `@` can run as formulas in Excel. Neutralize them when exporting untrusted data:

```python
def sanitize_cell(v: str) -> str:
    return "'" + v if v and v[0] in "=+-@" else v
```

## 6.6 Safe serialization practices

```python
# 1. Prefer JSON for untrusted or interchange data
data = json.loads(untrusted_text)

# 2. NEVER unpickle untrusted data. It can execute arbitrary code.
import pickle
# pickle.loads(untrusted_bytes)   # UNSAFE: remote code execution risk

# 3. Use yaml.safe_load, never yaml.load
import yaml  # pip install pyyaml
cfg = yaml.safe_load(open("c.yaml", encoding="utf-8"))

# 4. Validate the structure after parsing (dataclass example)
from dataclasses import dataclass

@dataclass(frozen=True)
class User:
    name: str
    age: int

def parse_user(raw: dict) -> User:
    if not isinstance(raw.get("name"), str) or not isinstance(raw.get("age"), int):
        raise ValueError("Invalid user payload")
    if not 0 <= raw["age"] <= 150:
        raise ValueError("Age out of range")
    return User(raw["name"], raw["age"])

# 5. Limit input size before parsing, to avoid denial of service
MAX = 1_000_000
raw = open("in.json", "rb").read(MAX + 1)
if len(raw) > MAX:
    raise ValueError("Payload too large")
user = parse_user(json.loads(raw))

# 6. Pydantic for schema validation (popular third-party option)
# from pydantic import BaseModel
# class User(BaseModel): name: str; age: int
# User.model_validate_json(text)
```

**Summary of rules**
- Use `with` so files are always closed.
- Always set `encoding="utf-8"`, and `newline=""` for CSV.
- Resolve and check user-supplied paths against a base directory.
- Prefer JSON. Avoid `pickle`, and use `yaml.safe_load`.
- Validate parsed data and cap its size.
