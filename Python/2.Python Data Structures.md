# 2. Python Data Structures: Examples

## 2.1 Built-ins

**List** (ordered, mutable)
```python
nums = [3, 1, 2]
nums.append(4)          # [3, 1, 2, 4]
nums.insert(0, 9)       # [9, 3, 1, 2, 4]
nums.remove(1)          # removes first match
last = nums.pop()       # removes and returns 4
nums.sort()             # in place
print(nums[1:3], nums[::-1])   # slicing, reversed copy
```

**Tuple** (ordered, immutable)
```python
point = (3, 4)
x, y = point                    # unpacking
single = (5,)                   # trailing comma needed
first, *rest = (1, 2, 3, 4)     # rest = [2, 3, 4]
```

**Set** (unique, unordered)
```python
a = {1, 2, 3}
b = {3, 4}
a.add(5)
print(a | b)    # union
print(a & b)    # intersection
print(a - b)    # difference
print(a ^ b)    # symmetric difference
print(2 in a)   # O(1) membership
```

**Dict** (key-value, insertion-ordered)
```python
user = {"name": "Ada", "age": 30}
user["email"] = "ada@example.com"
print(user.get("phone", "n/a"))   # safe access with default
user.update({"age": 31})
del user["email"]
merged = user | {"role": "admin"} # 3.9+

# Ways to iterate a dict
for k in user: print(k)
for v in user.values(): print(v)
for k, v in user.items(): print(k, v)
```

**String** (immutable sequence)
```python
s = "  Hello, World  "
print(s.strip().lower())          # 'hello, world'
print(s.split(","))               # ['  Hello', ' World  ']
print("-".join(["a", "b", "c"]))  # 'a-b-c'
print(s.replace("World", "Py"))
print(s.strip().startswith("He"))
print(s[2:7])                     # slicing
```

## 2.2 Comprehensions and Generators

**Comprehensions**
```python
squares = [n * n for n in range(5)]                 # list
evens = [n for n in range(10) if n % 2 == 0]        # with filter
unique_len = {len(w) for w in ["a", "bb", "cc"]}    # set
sq_map = {n: n * n for n in range(4)}               # dict
pairs = [(i, j) for i in range(2) for j in range(2)]  # nested
```

**Generators** (lazy, memory efficient)
```python
gen_expr = (n * n for n in range(1_000_000))   # no list built
print(next(gen_expr))                          # 0

def countdown(n):
    while n > 0:
        yield n
        n -= 1

print(list(countdown(3)))   # [3, 2, 1]
print(sum(n for n in range(100)))
```

## 2.3 `collections`
```python
from collections import deque, Counter, defaultdict, namedtuple

# deque: O(1) at both ends
d = deque([1, 2, 3])
d.appendleft(0)
d.pop()
d.rotate(1)

# Counter: frequency counting
c = Counter("banana")
print(c.most_common(2))     # [('a', 3), ('n', 2)]

# defaultdict: auto-creates missing keys
groups = defaultdict(list)
for word in ["apple", "avocado", "banana"]:
    groups[word[0]].append(word)

# namedtuple: lightweight immutable record
Point = namedtuple("Point", ["x", "y"])
p = Point(1, 2)
print(p.x, p._asdict())
```

## 2.4 `heapq`, `bisect`, `array`
```python
import heapq, bisect
from array import array

# heapq: min-heap
h = []
heapq.heappush(h, 5)
heapq.heappush(h, 1)
heapq.heappush(h, 3)
print(heapq.heappop(h))             # 1
print(heapq.nlargest(2, [4, 9, 2])) # [9, 4]

# max-heap trick: negate values
heapq.heappush(h, -10)

# bisect: binary search on sorted list
data = [1, 3, 5, 7]
print(bisect.bisect_left(data, 5))  # 2
bisect.insort(data, 4)              # keeps sorted: [1, 3, 4, 5, 7]

# array: compact typed array
arr = array("i", [1, 2, 3])         # 'i' = signed int
arr.append(4)
print(arr.tolist())
```

**Linked list**
```python
class Node:
    def __init__(self, val, next=None):
        self.val = val
        self.next = next

class LinkedList:
    def __init__(self):
        self.head = None

    def prepend(self, val):
        self.head = Node(val, self.head)

    def __iter__(self):
        cur = self.head
        while cur:
            yield cur.val
            cur = cur.next

ll = LinkedList()
ll.prepend(2)
ll.prepend(1)
print(list(ll))   # [1, 2]
```

**Binary search tree**
```python
class TreeNode:
    def __init__(self, key):
        self.key = key
        self.left = self.right = None

def insert(root, key):
    if root is None:
        return TreeNode(key)
    if key < root.key:
        root.left = insert(root.left, key)
    else:
        root.right = insert(root.right, key)
    return root

def inorder(root):
    if root:
        yield from inorder(root.left)
        yield root.key
        yield from inorder(root.right)

root = None
for k in [5, 3, 8, 1]:
    root = insert(root, k)
print(list(inorder(root)))   # [1, 3, 5, 8]
```

**Hash table** (simplified, with chaining)
```python
class HashTable:
    def __init__(self, size=8):
        self.buckets = [[] for _ in range(size)]

    def _bucket(self, key):
        return self.buckets[hash(key) % len(self.buckets)]

    def put(self, key, value):
        bucket = self._bucket(key)
        for i, (k, _) in enumerate(bucket):
            if k == key:
                bucket[i] = (key, value)
                return
        bucket.append((key, value))

    def get(self, key, default=None):
        for k, v in self._bucket(key):
            if k == key:
                return v
        return default

ht = HashTable()
ht.put("a", 1)
print(ht.get("a"))   # 1
```
